Woods Hole

Cape Cod and WHOI-inspired palettes

Originally published as PaletteWoodsHole as an R package. Hand-picked color palettes inspired by Woods Hole, Falmouth, and Woods Hole Oceanographic Institution. Eel Pond in winter, Siders Pond at sunset, the Shining Sea Bikeway, and more. This mode browses all 20 palettes and gives you a copy-ready hex list for whichever one fits your figure.

# install the source package
remotes::install_github("shu251/PaletteWoodsHole")
library(PaletteWoodsHole)
show_color(atlantis)   # scales::show_col() grid, as used in this package

library(tidyverse)
library(glue)

woods_hole <- read_csv("../data/woods-hole-palettes.csv")

swatch_html <- function(hex, cap = 8) {
  n <- length(hex)
  shown <- head(hex, cap)
  spans <- glue('<span class="mini-swatch" style="background:{shown}"></span>')
  out <- paste(spans, collapse = "")
  if (n > cap) out <- paste0(out, glue('<span class="palette-module-more">+{n - cap} more</span>'))
  out
}

gallery <- woods_hole |>
  arrange(palette, position) |>
  group_by(palette, display_name, description) |>
  summarise(
    n_colors = n(),
    swatches = swatch_html(hex),
    .groups = "drop"
  ) |>
  arrange(display_name)

cat(
  '<div class="palette-gallery">',
  glue_data(
    gallery,
    '<div class="palette-module">
  <h4>{display_name} <span class="palette-module-count">{n_colors} colors</span></h4>
  <p class="palette-module-desc">{description}</p>
  <div class="palette-module-swatches">{swatches}</div>
</div>'
  ),
  '</div>',
  sep = "\n"
)

whData = FileAttachment("../data/woods-hole-palettes.csv").csv({typed: true})
whPaletteMeta = Array.from(
  d3.rollup(whData, v => v[0].display_name, d => d.palette),
  ([palette, display_name]) => ({palette, display_name})
).sort((a, b) => a.display_name.localeCompare(b.display_name))
whOptions = new Map(whPaletteMeta.map(d => [d.display_name, d.palette]))

function whTextColor(hex) {
  const c = hex.replace("#", "");
  const r = parseInt(c.substring(0, 2), 16);
  const g = parseInt(c.substring(2, 4), 16);
  const b = parseInt(c.substring(4, 6), 16);
  const lum = (0.299 * r + 0.587 * g + 0.114 * b) / 255;
  return lum > 0.6 ? "#241C1C" : "#FFFFFF";
}
