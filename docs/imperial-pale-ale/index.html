IPA

Color palettes from craft beer brewing

IPAs are hoppy brews with rich amber and copper tones. I’m always looking for the best West Coast IPAs and they usually come with great art. Use these palettes to represent color schemes I’ve seen on cans, bottles, and more.

library(tidyverse)
library(glue)

imperial_ale <- read_csv("../data/imperial-pale-ale-colors.csv")

swatch_html <- function(hex, name, cap = 8) {
  n <- length(hex)
  hex <- head(hex, cap)
  name <- head(name, cap)
  spans <- glue('<span class="mini-swatch" style="background:{hex}" title="{name}"></span>')
  out <- paste(spans, collapse = "")
  if (n > cap) out <- paste0(out, glue('<span class="palette-module-more">+{n - cap} more</span>'))
  out
}

gallery <- imperial_ale |>
  group_by(palette) |>
  summarise(
    n_colors = n(),
    palette_name = first(display_name),
    examples = paste(head(hex, 4), collapse = ", "),
    swatches = swatch_html(hex, display_name),
    .groups = "drop"
  ) |>
  arrange(palette)

cat(
  '<div class="palette-gallery">',
  glue_data(
    gallery,
    '<div class="palette-module">
  <h4>{palette_name} <span class="palette-module-count">{n_colors} colors</span></h4>
  <p class="palette-module-desc">e.g. {examples}</p>
  <div class="palette-module-swatches">{swatches}</div>
</div>'
  ),
  '</div>',
  sep = "\n"
)

How to use it

  1. Pick a palette below.
  2. Click Randomize to shuffle the colors.
  3. Copy the R vector — or the plain hex list — straight into your plot.

function textColor(hex) {
  const c = hex.replace("#", "");
  const r = parseInt(c.substring(0, 2), 16);
  const g = parseInt(c.substring(2, 4), 16);
  const b = parseInt(c.substring(4, 6), 16);
  const lum = (0.299 * r + 0.587 * g + 0.114 * b) / 255;
  return lum > 0.6 ? "#241C1C" : "#FFFFFF";
}

vectorCopyBtn = {
  const label = () => `Copy R vector (${sample.length})`;
  const btn = html`<button class="swatch-copy-btn">${label()}</button>`;
  btn.onclick = () => {
    navigator.clipboard.writeText(hexVector).then(() => {
      btn.textContent = "Copied!";
      setTimeout(() => { btn.textContent = label(); }, 1200);
    });
  };
  return btn;
}
